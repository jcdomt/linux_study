### 输入子系统

#### 基础信息

输入子系统是 Linux 为输入事件准备的框架，像键盘、鼠标、触摸屏这些都是输入设备，但这些输入设备的类型都不一样，所以为了统一这些为了这些输入设备的驱动标准应运而生的

所有输入设备的节点会挂载在 `/dev/input` 

查看当前系统有哪些输入设备

```shell
cat /proc/bus/input/devices
```

#### 代码定义

我们上报的数据要按照具体格式要求上报到输入子系统的核心层，我们的应用就可以通过设备节点来获取输入信息

输入事件定义在`include/uapi/index/input.h`的 `input_event` 结构体中

```C
struct input_event {
#if (__BITS_PER_LONG != 32 || !defined(__USE_TIME_BITS64)) && !defined(__KERNEL__)
    // 事件发生的事件
	struct timeval time;
#define input_event_sec time.tv_sec
#define input_event_usec time.tv_usec
#else
	__kernel_ulong_t __sec;
#if defined(__sparc__) && defined(__arch64__)
	unsigned int __usec;
	unsigned int __pad;
#else
	__kernel_ulong_t __usec;
#endif
#define input_event_sec  __sec
#define input_event_usec __usec
#endif
    // 事件的类型
	__u16 type;
	__u16 code;
	__s32 value;
};

```

其中对于 `type` 类型，Linux 定义了以下值：
```C
#define EV_SYN			0x00	// 同步事件
#define EV_KEY			0x01	// 按键事件
#define EV_REL			0x02	// 相对坐标事件
#define EV_ABS			0x03	// 绝对坐标事件
#define EV_MSC			0x04	// 杂项（其他）事件
#define EV_SW			0x05	// 开关事件
#define EV_LED			0x11	// LED
#define EV_SND			0x12	// 声音事件
#define EV_REP			0x14	// 重复事件
#define EV_FF			0x15	// 压力事件
#define EV_PWR			0x16	// 电源事件
#define EV_FF_STATUS		0x17// 压力状态事件
#define EV_MAX			0x1f
#define EV_CNT			(EV_MAX+1)
```

例如当 `type == EV_KEY` 时，`code`被定义为：
```C
#define KEY_RESERVED		0
#define KEY_ESC			1
#define KEY_1			2
#define KEY_2			3
#define KEY_3			4
#define KEY_4			5
#define KEY_5			6
#define KEY_6			7
#define KEY_7			8
#define KEY_8			9
#define KEY_9			10
#define KEY_0			11
#define KEY_MINUS		12
#define KEY_EQUAL		13
#define KEY_BACKSPACE		14
#define KEY_TAB			15
#define KEY_Q			16
// ...
```

此时，`value`为0表示抬起，如果为1表示按下，如果是2表示长按

#### 用户端程序

读取 `event0` 的事件，并通过 `struct input_event` 读取参数

```C
#include <linux/input.h>
#include <fcntl.h>
#include <stdio.h>
#include <unistd.h>

int main(int argc, char **argv)
{
    int fd = open("/dev/input/event0", O_RDONLY);
    if (fd < 0) {
        perror("Failed to open input device");
        return 1;
    }

    struct input_event ev;
    while (1) {
        read(fd, &ev, sizeof(struct input_event));
        printf("Event: time %ld.%06ld, type %d, code %d, value %d\n",
               ev.input_event_sec, ev.input_event_usec,
               ev.type, ev.code, ev.value);
    }

    return 0;
}
```

#### 驱动端程序

我们可以使用开发板上的按键值设置为 `input.h` 文件中宏定义中的任意一个，比如将开发板上的 KEY 按键设置为 `KEY_0`

#### 输入子系统的API

###### 1. `input_allocate_device` 函数

在编写 `input` 设备驱动的时候需要先申请一个 `input_dev` 结构体变量，使用 `input_allocate_device` 来申请

```C
struct input_dev *input_allocate_device(void);
// 返回申请到的 input_dev
```

###### 2. `input_free_device` 函数

```C
// 申请到的 input_dev 结构体
void input_free_device(struct input_dev *dev); 
```

###### 3. `__set_bit` 函数 和 `input_dev` 的初始化

```C
struct input_dev dev = input_allocate_device();
// 设置名称
dev->name = "name";
// 设置输入事件的类型
__set_bit(SV_KEY, dev->evbit);
// 设置对应的参数
__set_bit(KEY_A, dev->keybit);
```

###### 4. `input_register_device` 函数

用于注册

```C
int ret = input_register_device(dev);
```

###### 5. `input_unregister_device`函数

```C
input_unregister_device(struct input_dev dev);
```

###### 6. `input_report_xxx` 函数

最终我们需要把时间上报上去，上报时间我们使用的函数要针对具体的时间来上报。比如，按键我们使用`input_report_key`函数，同样的还有一些其他的事件上报函数

```C
void input_report_rel(struct input_dev *dev, unsigned int code, int value);
void input_report_abs(struct input_dev *dev, unsigned int code, int value);
void input_report_ff_state(struct input_dev *dev, unsigned int code, int value);
// ...
```

###### 7. `input_sync` 函数

当上报时间后还需要通过`input_sync`函数通知内核告诉 `Linux` 内核 `input`子系统上报结束

```C
void input_sync(struct input_dev *dev);
```

#### 结合按键程序进行事件通报

我们可以结合消抖的按键程序

```C
// 使用内核定时器进行按键消抖

#include <linux/module.h>
MODULE_LICENSE("GPL");

#include <linux/platform_device.h>
#include <linux/of.h>
#include <linux/of_gpio.h>

#include <linux/interrupt.h>

#include <linux/timer.h>

#include <uapi/linux/input.h>
#include <linux/input.h>


void key_shake_timer_handler(struct timer_list *t);

struct timer_data {
    struct timer_list timer;
    int gpio;
    int irq;
    struct platform_device *pdev;
    struct input_dev *input_dev;
    
    // 记录 GPIO 按键刚按下时的状态
    int origin_key_state;
    int stable_key_state
};

void key_shake_timer_handler(struct timer_list *t)
{
    printk(KERN_INFO "Key shake timer handler called\n");
    // 从 timer_list 中获取我们的数据结构
    struct timer_data *data = from_timer(data, t, timer);
    // 继续读取 GPIO 状态，判断按键是否仍然按下
    int key_state = gpio_get_value(data->gpio);
    if (key_state == data->origin_key_state && key_state != data->stable_key_state) {
        printk(KERN_INFO "Key is still pressed after debounce\n");
        // 这里可以认为按键有效，向输入子系统报告按键事件
        input_report_key(data->input_dev, KEY_A, !key_state);
        input_sync(data->input_dev);
        data->stable_key_state = key_state;
    } else {
        printk(KERN_INFO "Key is released after debounce\n");  
    }
}

irqreturn_t key_irq_handler(int irq, void *dev_id)
{
    printk(KERN_INFO "Key IRQ handler called for IRQ %d\n", irq);

    // 我们在这里进行消抖，如果 20ms 后按键状态仍然为按下，则认为按键有效
    // 使用内核定时器
    struct timer_data *data = (struct timer_data *)dev_id;
    // 记录按键刚按下时的状态
    data->origin_key_state = gpio_get_value(data->gpio);
    mod_timer(&data->timer, jiffies + msecs_to_jiffies(20));
    return IRQ_HANDLED;
}

int key_probe(struct platform_device *pdev)
{
    printk(KERN_INFO "Key driver probe function called\n");
    
    struct device_node *node = pdev->dev.of_node;
    if (!node) {
        printk(KERN_ERR "Failed to get device node\n");
        return -ENODEV;
    }

    // 申请 GPIO 资源
    int key_gpio = of_get_named_gpio(node, "key-gpios", 0);
    if (key_gpio < 0) {
        printk(KERN_ERR "Failed to get key GPIO from device tree\n");
        return -EINVAL;
    }
    int ret = gpio_request(key_gpio, "key_gpio");
    if (ret) {
        printk(KERN_ERR "Failed to request GPIO %d\n", key_gpio);
        return -EIO;
    }

    // 配置 GPIO 为输入模式
    ret = gpio_direction_input(key_gpio);
    if (ret) {
        printk(KERN_ERR "Failed to set GPIO %d as input\n", key_gpio);
        goto gpio_free;
    }
    ret = gpio_set_debounce(key_gpio, 200); // 设置消抖时间为 200ms
    if (ret) {
        printk(KERN_ERR "Failed to set debounce for GPIO %d\n", key_gpio);
        goto gpio_free;
    }

    // 从 GPIO 申请中断
    int irq = gpio_to_irq(key_gpio);
    if (irq < 0) {
        printk(KERN_ERR "Failed to get IRQ for GPIO %d\n", key_gpio);
        goto gpio_free;
    }

    // 组成 timer_data 结构体
    // 以设备的名义申请一块内存，内核会在设备移除时自动释放
    struct timer_data *data = devm_kzalloc(&pdev->dev, sizeof(struct timer_data), GFP_KERNEL);
    if (!data) {
        printk(KERN_ERR "Failed to allocate memory for timer_data\n");
        goto gpio_free;
    }
    data->gpio = key_gpio;
    data->irq = irq;
    data->pdev = pdev;
    timer_setup(&data->timer, key_shake_timer_handler, 0);

    ret = request_irq(irq, key_irq_handler, IRQF_TRIGGER_FALLING | IRQF_TRIGGER_RISING, "key_irq", data);
    if (ret) {
        printk(KERN_ERR "Failed to request IRQ %d\n", irq);
        goto irq_free;
    }

    // 向内核申请输入设备
    data->input_dev = input_allocate_device();
    if (!data->input_dev) {
        printk(KERN_ERR "Failed to allocate input device\n");
        ret = -ENOMEM;
        goto timer_free;
    }
    data->input_dev->name = "key_input_device";
    set_bit(EV_KEY, data->input_dev->evbit);
    set_bit(KEY_A, data->input_dev->keybit); // 假设按键对应 KEY_A
    ret = input_register_device(data->input_dev);
    if (ret) {
        printk(KERN_ERR "Failed to register input device\n");
        goto input_dev_free;
    }

    platform_set_drvdata(pdev, data);

    return 0;
input_dev_free:
    input_free_device(data->input_dev);
    input_unregister_device(data->input_dev);
timer_free:
    del_timer_sync(&data->timer);
irq_free:
    free_irq(key_gpio, data);
gpio_free:
    gpio_free(key_gpio);
    return ret;
}

int key_remove(struct platform_device *pdev)
{
    printk(KERN_INFO "Key driver remove function called\n");
    struct timer_data *data = dev_get_drvdata(&pdev->dev);
    del_timer_sync(&data->timer);

    if (data) {
        free_irq(data->irq, data);
        gpio_free(data->gpio);
        input_unregister_device(data->input_dev);
        input_free_device(data->input_dev);
    }

    return 0;
}

struct platform_driver key_driver = {
    .probe = key_probe,
    .remove = key_remove,
    .driver = {
        .owner = THIS_MODULE,
        .name = "key_driver",
        .of_match_table = (const struct of_device_id[]) {
            { .compatible = "test, key" },
            { /* sentinel */ }
        },
    },
};

static int __init key_driver_init(void)
{
    return platform_driver_register(&key_driver);
}

static void __exit key_driver_exit(void)
{
    platform_driver_unregister(&key_driver);
}

module_init(key_driver_init);
module_exit(key_driver_exit);
```
