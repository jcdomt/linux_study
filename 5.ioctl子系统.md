### `ioctl`子系统

#### `unlocked_ioctl`接口 和 `read/write`函数的相同点和不同点



|                  | 相同点               | 不同点                                                       |
| ---------------- | -------------------- | ------------------------------------------------------------ |
| `ioctl`系统      | 都可以往内核读写数据 | 既可以读也可以写<br />但是读取大数据时效率不高               |
| `read/write`函数 | -                    | 需要两个函数，一个函数负责读，一个函数负责写<br />读取大数据时效率高 |

##### `ioctl`接口命令规则

| 命令分区 | 字节数  | 功能               | 备注                                                         |
| -------- | ------- | ------------------ | ------------------------------------------------------------ |
| 1        | 0 - 7   | 命令的编号         | 范围：0 - 255                                                |
| 2        | 8 - 15  | 命令的幻数         |                                                              |
| 3        | 16 - 29 | 表示传递的数据大小 |                                                              |
| 4        | 30 - 31 | 表示读写的方向     | 00：表示用户程序和驱动程序没有数据传递<br />10：表示用户程序向驱动读数据<br />01：表示用户程序向驱动写数据<br />11：表示先向驱动写数据，然后从驱动里把数据读出来 |

第一个分区和第二个分区的主要作用是区分命令

#### 命令的合成宏和分解宏

##### 合成宏

- type: 表示命令组成的魔数
- nr: 定义命令组成的编号
- size: 表示命令组成的参数传递大小，注意这里不是传递数字，而是数据类型，如传递 4字节，就可以写成 `int`

| 宏                      | 功能                         |
| ----------------------- | ---------------------------- |
| `_IO(type, nr)`         | 用来定义没有数据传递的命令   |
| `_IOR(type, nr, size)`  | 用来定义向驱动读数据的命令   |
| `_IOW(type, nr, size)`  | 用来定义向驱动写数据的命令   |
| `_IOWR(type, nr, size)` | 用来定义向驱动先写后读的命令 |

##### 分解宏

| 宏              | 功能                   |
| --------------- | ---------------------- |
| `_IOC_DIR(nr)`  | 分解命令的方向         |
| `_IOC_TYPE(nr)` | 分解命令的魔数         |
| `_IOC_NR(nr)`   | 分解命令的编号         |
| `_IOC_SIZE(nr)` | 分解命令复制的数据大小 |

 

#### `ioctl`接口

 `ioctl`定义在`file_operation`中的`unlocked_ioctl`，定义如下：

```C
// cmd: 完整的 32 位 IO 指令
// value: 传入的值
long (*unlocked_ioctl) (struct file *file, unsigned int command, unsigned long value);
```

在 `app`应用中使用`ioctl`

```c
#include <stdio.h>
#include <fcntl.h>      // open(), O_RDWR
#include <unistd.h>     // close(), read(), write()
#include <sys/ioctl.h>  // ioctl()

#define CMD_TEST_O _IO('T', 0)
#define CMD_TEST_I _IOW('T', 1, int) 
#define CMD_TEST_R _IOR('T', 2, int)

int main(int argc, char *argv[])
{
    int fd = open("/dev/ioctl_test", O_RDWR);
    if (fd < 0) {
        perror("Failed to open device");
        return -1;
    }


    ioctl(fd, CMD_TEST_O); // 调用无参数的 ioctl 命令
    ioctl(fd, CMD_TEST_I, 42); // 调用带参数的 ioctl 命令，传递整数 42
    int value;
    ioctl(fd ,CMD_TEST_R, &value); // 调用读取参数的 ioctl 命令，获取整数值
    printf("Value read from ioctl: %d\n", value);
    
    close(fd);
    return 0;
}

```

在内核中实现`ioctl`

```C
#include <linux/module.h>

#include <linux/ioctl.h>
#include <linux/miscdevice.h>

MODULE_LICENSE("GPL");

#define CMD_TEST_O _IO('T', 0)
#define CMD_TEST_I _IOW('T', 1, int) 
#define CMD_TEST_R _IOR('T', 2, int)

// 使用最开始的 misc_device 来尝试 ioctl

static long ioctl_handler(struct file *file, unsigned int cmd, unsigned long arg)
{
    switch (cmd) {
        case CMD_TEST_O:
            printk(KERN_INFO "CMD_TEST_O received\n");
            break;
        case CMD_TEST_I:
            printk(KERN_INFO "CMD_TEST_I received with value: %ld\n", arg);
            break;
        case CMD_TEST_R:
            {
                int value = 123; // 假设我们要返回的值
                if (copy_to_user((int __user *)arg, &value, sizeof(value))) {
                    return -EFAULT;
                }
                printk(KERN_INFO "CMD_TEST_R received, returning value: %d\n", value);
            }
            break;
        default:
            return -EINVAL;
    }
    return 0;
}

static struct file_operations ioctl_fops = {
    .owner = THIS_MODULE,
    .unlocked_ioctl = ioctl_handler, // 这里可以实现 ioctl 的处理函数
};

static struct miscdevice ioctl_misc_device = {
    .minor = MISC_DYNAMIC_MINOR,
    .name = "ioctl_test",
    .fops = &ioctl_fops,
};

static int __init ioctl_init(void)
{
    printk(KERN_INFO "IOCTL module loaded\n");

    // 创建 misc 设备
    int ret = misc_register(&ioctl_misc_device);
    if (ret < 0) {
        printk(KERN_ERR "Failed to register misc device\n");
        return ret;
    }

    return 0;
}

static void __exit ioctl_exit(void)
{
    printk(KERN_INFO "IOCTL module unloaded\n");

    // 注销 misc 设备
    misc_deregister(&ioctl_misc_device);
}

module_init(ioctl_init);
module_exit(ioctl_exit);
```
